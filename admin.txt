import React from 'react';
import { useAppStore } from '../store';
import { Navigate } from 'react-router-dom';
import { Users, PackageOpen, DollarSign, Activity, X, Tag, Trash2 } from 'lucide-react';
import { useState } from 'react';
import { ImageUpload } from '../components/ImageUpload';
import { MultiImageUpload } from '../components/MultiImageUpload';
import { FileUpload } from "../components/FileUpload";
import { collection, addDoc, deleteDoc, doc, setDoc } from "firebase/firestore";
import { db } from "../lib/firebase";
import { formatPrice } from '../lib/currency';

export function Admin() {
  const { user, products, addProduct, removeProduct, coupons, addCoupon, removeCoupon } = useAppStore();
  
  const [isAddProductOpen, setIsAddProductOpen] = useState(false);
  const [isManageProductsOpen, setIsManageProductsOpen] = useState(false);
  const [isCouponModalOpen, setIsCouponModalOpen] = useState(false);
  const [isAnalyticsModalOpen, setIsAnalyticsModalOpen] = useState(false);
  
  const [newProduct, setNewProduct] = useState({
    title: '',
    description: '',
    price: 0,
    category: 'cc',
    creator: '',
    imageUrl: '',
    images: [] as string[],
    beforeImageUrl: '',
    afterImageUrl: '',
    features: '',
    fileUrl: ''
  });

  const [newCoupon, setNewCoupon] = useState({
    code: '',
    discountPercentage: 10
  });

  if (!user) {
    return (
      <div className="container mx-auto px-4 py-20 text-center">
        <h2 className="text-2xl font-bold text-error">Access Denied</h2>
        <p className="text-text-secondary mt-2">Please log in to view this page.</p>
      </div>
    );
  }

  const isAdmin = user?.email === 'animerepear620@gmail.com' || user?.email === 'nikhilcoder2222@gmail.com';

  if (!isAdmin) {
    return (
      <div className="container mx-auto px-4 py-20 text-center">
        <h2 className="text-2xl font-bold text-error">Access Denied</h2>
        <p className="text-text-secondary mt-2">You do not have permission to view this page.</p>
      </div>
    );
  }

  const handleAddProduct = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!isAdmin) return;
    
    if (!newProduct.fileUrl) {
      alert("Please upload the product file/pack before saving.");
      return;
    }
    
    const productId = Math.random().toString(36).substr(2, 9);
    
    const productData = {
      id: productId,
      title: newProduct.title,
      description: newProduct.description,
      price: Number(newProduct.price),
      category: newProduct.category,
      creator: newProduct.creator || undefined,
      imageUrl: newProduct.imageUrl || newProduct.afterImageUrl, // fallback if cc
      images: newProduct.images,
      beforeImageUrl: newProduct.category === 'cc' ? newProduct.beforeImageUrl : undefined,
      afterImageUrl: newProduct.category === 'cc' ? newProduct.afterImageUrl : undefined,
      features: newProduct.features.split(',').map(f => f.trim()).filter(Boolean),
      fileUrl: newProduct.fileUrl
    };

    try {
      await setDoc(doc(db, 'products', productId), productData);
      addProduct(productData);
      
      setIsAddProductOpen(false);
      setNewProduct({ title: '', description: '', price: 0, category: 'cc', creator: '', imageUrl: '', images: [], beforeImageUrl: '', afterImageUrl: '', features: '', fileUrl: '' });
      alert("Product added successfully!");
    } catch (error) {
      console.error("Error adding product:", error);
      alert("Failed to add product. Please try again.");
    }
  };

  const handleAddCoupon = (e: React.FormEvent) => {
    e.preventDefault();
    if (!isAdmin) return;
    
    addCoupon({
      id: Math.random().toString(36).substr(2, 9),
      code: newCoupon.code.toUpperCase(),
      discountPercentage: Number(newCoupon.discountPercentage)
    });
    
    setIsCouponModalOpen(false);
    setNewCoupon({ code: '', discountPercentage: 10 });
    alert("Coupon added successfully!");
  };

  const stats = [
    { label: 'Total Revenue', value: formatPrice(12450), icon: DollarSign, trend: '+14%' },
    { label: 'Active Users', value: '842', icon: Users, trend: '+5%' },
    { label: 'Products', value: products.length.toString(), icon: PackageOpen, trend: 'Total' },
    { label: 'Active Coupons', value: coupons.length.toString(), icon: Tag, trend: 'Active' },
  ];

  return (
    <div className="container mx-auto px-4 py-12">
      <div className="flex items-center justify-between mb-8 pb-4 border-b border-border">
        <h1 className="text-3xl font-bold">Admin Dashboard</h1>
        <div className="bg-primary/10 text-primary px-3 py-1 rounded-full text-sm font-medium">
          {isAdmin ? 'Admin Mode' : 'Demo Mode (Read Only)'}
        </div>
      </div>
      
      {!isAdmin && (
        <div className="bg-secondary/10 border border-secondary/20 text-secondary px-4 py-3 rounded-xl mb-8 flex items-center">
          <DollarSign className="w-5 h-5 mr-3 shrink-0" />
          <p className="text-sm">You are viewing this dashboard in demo mode. Actions like adding products or creating coupons are disabled.</p>
        </div>
      )}

      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-6 mb-12">
        {stats.map((stat, i) => {
          const Icon = stat.icon;
          return (
            <div key={i} className="bg-surface p-6 rounded-xl border border-border">
              <div className="flex items-center justify-between mb-4">
                <div className="p-2 bg-background rounded-lg text-primary">
                  <Icon className="w-6 h-6" />
                </div>
                <span className="text-success text-sm font-medium">{stat.trend}</span>
              </div>
              <p className="text-text-secondary text-sm font-medium">{stat.label}</p>
              <h3 className="text-3xl font-bold mt-1">{stat.value}</h3>
            </div>
          );
        })}
      </div>

      <div className="grid grid-cols-1 lg:grid-cols-3 gap-8">
        <div className="lg:col-span-2 bg-surface rounded-xl border border-border overflow-hidden">
          <div className="p-6 border-b border-border flex justify-between items-center">
            <h2 className="text-xl font-bold">Recent Orders</h2>
            <button className="text-primary text-sm font-medium hover:underline">View All</button>
          </div>
          <div className="overflow-x-auto">
            <table className="w-full text-left">
              <thead className="bg-background text-text-secondary text-sm">
                <tr>
                  <th className="px-6 py-3 font-medium">Order ID</th>
                  <th className="px-6 py-3 font-medium">Customer</th>
                  <th className="px-6 py-3 font-medium">Product</th>
                  <th className="px-6 py-3 font-medium">Amount</th>
                  <th className="px-6 py-3 font-medium">Status</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-border">
                {[1, 2, 3, 4].map((i) => (
                  <tr key={i} className="hover:bg-background/50 transition-colors">
                    <td className="px-6 py-4 text-sm font-medium">#FW-{1000 + i}</td>
                    <td className="px-6 py-4 text-sm text-text-secondary">user{i}@example.com</td>
                    <td className="px-6 py-4 text-sm">Cinematic CC Pack</td>
                    <td className="px-6 py-4 text-sm font-medium">${(29.99 * i).toFixed(2)}</td>
                    <td className="px-6 py-4">
                      <span className="bg-success/10 text-success text-xs px-2 py-1 rounded-full font-medium">
                        Completed
                      </span>
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        </div>

        <div className="bg-surface rounded-xl border border-border p-6">
          <h2 className="text-xl font-bold mb-6">Quick Actions</h2>
          <div className="space-y-3">
            <button 
              onClick={() => isAdmin && setIsAddProductOpen(true)}
              className="w-full text-left px-4 py-3 bg-background hover:bg-border/50 border border-border rounded-lg font-medium transition-colors"
            >
              + Add New Product
            </button>
            <button 
              onClick={() => isAdmin && setIsManageProductsOpen(true)}
              className="w-full text-left px-4 py-3 bg-background hover:bg-border/50 border border-border rounded-lg font-medium transition-colors"
            >
              📦 Manage Products
            </button>
            <button 
              onClick={() => isAdmin && setIsCouponModalOpen(true)} 
              className="w-full text-left px-4 py-3 bg-background hover:bg-border/50 border border-border rounded-lg font-medium transition-colors"
            >
              % Create Coupon Code
            </button>
            <button 
              onClick={() => isAdmin && setIsAnalyticsModalOpen(true)} 
              className="w-full text-left px-4 py-3 bg-background hover:bg-border/50 border border-border rounded-lg font-medium transition-colors"
            >
              📊 View Analytics Report
            </button>
          </div>
        </div>
      </div>

      {/* Add Product Modal */}
      {isAddProductOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/60 backdrop-blur-sm">
          <div className="bg-surface border border-border w-full max-w-lg rounded-2xl shadow-2xl overflow-hidden relative">
            <button 
              onClick={() => setIsAddProductOpen(false)}
              className="absolute top-4 right-4 p-2 text-text-secondary hover:text-text-primary rounded-full hover:bg-background transition-colors"
            >
              <X className="w-5 h-5" />
            </button>
            <div className="p-8">
              <h2 className="text-2xl font-bold mb-6">Add New Product</h2>
              <form onSubmit={handleAddProduct} className="space-y-4">
                <div>
                  <label className="block text-sm font-medium text-text-secondary mb-1">Title</label>
                  <input type="text" required className="w-full bg-background border border-border rounded-xl px-4 py-2" value={newProduct.title} onChange={e => setNewProduct({...newProduct, title: e.target.value})} />
                </div>
                <div>
                  <label className="block text-sm font-medium text-text-secondary mb-1">Description</label>
                  <textarea required className="w-full bg-background border border-border rounded-xl px-4 py-2" value={newProduct.description} onChange={e => setNewProduct({...newProduct, description: e.target.value})} />
                </div>
                <div>
                  <label className="block text-sm font-medium text-text-secondary mb-1">Creator</label>
                  <input type="text" className="w-full bg-background border border-border rounded-xl px-4 py-2" value={newProduct.creator} onChange={e => setNewProduct({...newProduct, creator: e.target.value})} placeholder="Optional: Creator Name" />
                </div>
                <div className="grid grid-cols-2 gap-4">
                  <div>
                    <label className="block text-sm font-medium text-text-secondary mb-1">Price</label>
                    <input type="number" step="0.01" required className="w-full bg-background border border-border rounded-xl px-4 py-2" value={newProduct.price} onChange={e => setNewProduct({...newProduct, price: parseFloat(e.target.value)})} />
                  </div>
                  <div>
                    <label className="block text-sm font-medium text-text-secondary mb-1">Category</label>
                    <select className="w-full bg-background border border-border rounded-xl px-4 py-2" value={newProduct.category} onChange={e => setNewProduct({...newProduct, category: e.target.value})}>
                      <option value="cc">CC (Colour Correction)</option>
                      <option value="thumbnails">Thumbnails</option>
                      <option value="templates">Templates</option>
                    </select>
                  </div>
                </div>
                {newProduct.category === 'cc' ? (
                  <div className="grid grid-cols-2 gap-4">
                    <ImageUpload 
                      label="Before Image" 
                      value={newProduct.beforeImageUrl} 
                      onChange={v => setNewProduct({...newProduct, beforeImageUrl: v})} 
                    />
                    <ImageUpload 
                      label="After Image" 
                      value={newProduct.afterImageUrl} 
                      onChange={v => setNewProduct({...newProduct, afterImageUrl: v})} 
                    />
                  </div>
                ) : (
                  <ImageUpload 
                    label="Product Image (Main)" 
                    value={newProduct.imageUrl} 
                    onChange={v => setNewProduct({...newProduct, imageUrl: v})} 
                  />
                )}
                
                <MultiImageUpload 
                  label="Additional Gallery Images" 
                  images={newProduct.images} 
                  onChange={v => setNewProduct({...newProduct, images: v})} 
                />

                <FileUpload
                  label="Product File/Pack URL (ZIP, etc.)"
                  value={newProduct.fileUrl}
                  onChange={v => setNewProduct({...newProduct, fileUrl: v})}
                  required={true}
                />

                <div>
                  <label className="block text-sm font-medium text-text-secondary mb-1">Features (comma separated)</label>
                  <input type="text" className="w-full bg-background border border-border rounded-xl px-4 py-2" value={newProduct.features} onChange={e => setNewProduct({...newProduct, features: e.target.value})} placeholder="e.g. 4K, 60fps" />
                </div>
                <button type="submit" className="w-full bg-primary hover:bg-button-hover text-white py-3 rounded-xl font-bold mt-4 transition-colors">
                  Save Product
                </button>
              </form>
            </div>
          </div>
        </div>
      )}

      {/* Manage Products Modal */}
      {isManageProductsOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/60 backdrop-blur-sm">
          <div className="bg-surface border border-border w-full max-w-2xl rounded-2xl shadow-2xl overflow-hidden relative max-h-[80vh] flex flex-col">
            <button 
              onClick={() => setIsManageProductsOpen(false)}
              className="absolute top-4 right-4 p-2 text-text-secondary hover:text-text-primary rounded-full hover:bg-background transition-colors z-10"
            >
              <X className="w-5 h-5" />
            </button>
            <div className="p-8 flex-1 overflow-y-auto">
              <h2 className="text-2xl font-bold mb-6">Manage Products</h2>
              {products.length > 0 ? (
                <div className="space-y-3">
                  {products.map(product => (
                    <div key={product.id} className="flex flex-col sm:flex-row sm:items-center justify-between bg-background p-4 rounded-lg border border-border gap-4">
                      <div className="flex items-center gap-4">
                        <img src={product.beforeImageUrl || product.imageUrl} alt={product.title} className="w-12 h-12 rounded object-cover" />
                        <div>
                          <span className="font-bold block">{product.title}</span>
                          <span className="text-sm text-text-secondary capitalize">{product.category === 'cc' ? 'CC (Colour Correction)' : product.category} • {formatPrice(product.price)}</span>
                        </div>
                      </div>
                      <button 
                        onClick={async () => {
                          if (window.confirm('Are you sure you want to delete this product?')) {
                            try {
                              await deleteDoc(doc(db, 'products', product.id));
                              removeProduct(product.id);
                            } catch (error) {
                              console.error('Error deleting product:', error);
                              alert('Failed to delete product.');
                            }
                          }
                        }}
                        className="p-2 text-error hover:bg-error/10 rounded-lg transition-colors flex items-center gap-2 text-sm font-medium"
                      >
                        <Trash2 className="w-4 h-4" /> <span className="sm:hidden">Delete</span>
                      </button>
                    </div>
                  ))}
                </div>
              ) : (
                <p className="text-text-secondary text-center py-8">No products found.</p>
              )}
            </div>
          </div>
        </div>
      )}

      {/* Coupon Modal */}
      {isCouponModalOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/60 backdrop-blur-sm">
          <div className="bg-surface border border-border w-full max-w-md rounded-2xl shadow-2xl overflow-hidden relative">
            <button 
              onClick={() => setIsCouponModalOpen(false)}
              className="absolute top-4 right-4 p-2 text-text-secondary hover:text-text-primary rounded-full hover:bg-background transition-colors"
            >
              <X className="w-5 h-5" />
            </button>
            <div className="p-8">
              <h2 className="text-2xl font-bold mb-6">Create Coupon Code</h2>
              <form onSubmit={handleAddCoupon} className="space-y-4">
                <div>
                  <label className="block text-sm font-medium text-text-secondary mb-1">Coupon Code</label>
                  <input type="text" required className="w-full bg-background border border-border rounded-xl px-4 py-2 uppercase" value={newCoupon.code} onChange={e => setNewCoupon({...newCoupon, code: e.target.value.toUpperCase()})} placeholder="e.g. SUMMER25" />
                </div>
                <div>
                  <label className="block text-sm font-medium text-text-secondary mb-1">Discount Percentage</label>
                  <input type="number" min="1" max="100" required className="w-full bg-background border border-border rounded-xl px-4 py-2" value={newCoupon.discountPercentage} onChange={e => setNewCoupon({...newCoupon, discountPercentage: Number(e.target.value)})} />
                </div>
                <button type="submit" className="w-full bg-primary hover:bg-button-hover text-white py-3 rounded-xl font-bold mt-4 transition-colors">
                  Save Coupon
                </button>
              </form>
              
              {coupons.length > 0 && (
                <div className="mt-8 border-t border-border pt-4">
                  <h3 className="text-lg font-bold mb-4">Active Coupons</h3>
                  <div className="space-y-2 max-h-40 overflow-y-auto">
                    {coupons.map(coupon => (
                      <div key={coupon.id} className="flex items-center justify-between bg-background p-3 rounded-lg border border-border">
                        <div>
                          <span className="font-bold">{coupon.code}</span>
                          <span className="text-sm text-text-secondary ml-2">{coupon.discountPercentage}% OFF</span>
                        </div>
                        <button 
                          onClick={() => removeCoupon(coupon.id)} 
                          className="p-2 text-text-secondary hover:text-error hover:bg-error/10 rounded-lg transition-colors"
                          aria-label="Remove coupon"
                        >
                          <Trash2 className="w-5 h-5" />
                        </button>
                      </div>
                    ))}
                  </div>
                </div>
              )}
            </div>
          </div>
        </div>
      )}

      {/* Analytics Modal */}
      {isAnalyticsModalOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4 bg-black/60 backdrop-blur-sm">
          <div className="bg-surface border border-border w-full max-w-3xl rounded-2xl shadow-2xl overflow-hidden relative">
            <button 
              onClick={() => setIsAnalyticsModalOpen(false)}
              className="absolute top-4 right-4 p-2 text-text-secondary hover:text-text-primary rounded-full hover:bg-background transition-colors"
            >
              <X className="w-5 h-5" />
            </button>
            <div className="p-8">
              <h2 className="text-2xl font-bold mb-6">Analytics Report</h2>
              <div className="grid grid-cols-1 md:grid-cols-3 gap-6 mb-8">
                <div className="bg-background border border-border p-4 rounded-xl">
                  <p className="text-sm text-text-secondary mb-1">Conversion Rate</p>
                  <p className="text-2xl font-bold text-primary">3.2%</p>
                </div>
                <div className="bg-background border border-border p-4 rounded-xl">
                  <p className="text-sm text-text-secondary mb-1">Avg Order Value</p>
                  <p className="text-2xl font-bold text-primary">{formatPrice(42.50)}</p>
                </div>
                <div className="bg-background border border-border p-4 rounded-xl">
                  <p className="text-sm text-text-secondary mb-1">Refund Rate</p>
                  <p className="text-2xl font-bold text-error">0.8%</p>
                </div>
              </div>
              <div className="bg-background border border-border rounded-xl p-6 h-64 flex items-center justify-center relative overflow-hidden group">
                <div className="absolute inset-0 flex items-end">
                  <svg viewBox="0 0 100 50" preserveAspectRatio="none" className="w-full h-3/4 text-primary opacity-20">
                    <path d="M0,50 L0,40 Q10,30 20,35 T40,25 T60,15 T80,20 T100,5 L100,50 Z" fill="currentColor" />
                  </svg>
                  <svg viewBox="0 0 100 50" preserveAspectRatio="none" className="w-full h-full text-primary">
                    <path d="M0,50 L0,40 Q10,30 20,35 T40,25 T60,15 T80,20 T100,5" fill="none" stroke="currentColor" strokeWidth="2" />
                  </svg>
                </div>
                <div className="z-10 bg-surface/80 backdrop-blur px-4 py-2 rounded-lg border border-border shadow-sm hidden group-hover:block absolute">
                  Chart Data Simulated
                </div>
              </div>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}
